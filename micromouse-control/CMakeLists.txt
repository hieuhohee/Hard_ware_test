cmake_minimum_required(VERSION 3.20)
project(micromouse_control_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mm_core src/controller.cpp)
target_include_directories(mm_core PUBLIC include)
add_executable(control_tests tests/control_tests.cpp)
target_link_libraries(control_tests PRIVATE mm_core)
add_executable(platform_tests tests/platform_tests.cpp platform/board_config.cpp
    platform/stm32_platform.cpp platform/micromouse_app.cpp)
target_include_directories(platform_tests PRIVATE tests/hal_mock platform)
target_link_libraries(platform_tests PRIVATE mm_core)
foreach(target mm_core control_tests platform_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME controller_scenarios COMMAND control_tests)
add_test(NAME stm32_adapter_scenarios COMMAND platform_tests)
